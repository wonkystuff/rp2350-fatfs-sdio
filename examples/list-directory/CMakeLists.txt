cmake_minimum_required(VERSION 3.16)
if(NOT PICO_SDK_PATH)
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to your Pico SDK checkout")
endif()
set(PICO_BOARD pico2 CACHE STRING "RP2350 board")
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(sd_list_example LANGUAGES C CXX ASM)
pico_sdk_init()
add_subdirectory(../.. sd-card)
add_executable(sd_list main.cpp)
target_link_libraries(sd_list PRIVATE rp2350_fatfs_sdio::rp2350_fatfs_sdio)
pico_enable_stdio_usb(sd_list 1)
pico_enable_stdio_uart(sd_list 0)
pico_add_extra_outputs(sd_list)
